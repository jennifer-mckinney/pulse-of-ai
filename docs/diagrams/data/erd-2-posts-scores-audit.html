%% Summary: The immutable post, its three per-post scores, the decision audit log that ties every score to a registered methodology version, the embedding, and the job-level bias assessments.

erDiagram

  raw_posts {
    uuid id PK
    uuid source_id FK
    text external_id "UNIQUE with source_id; route:id, route:fp:HMAC, or route:SHA-256 when no PROVENANCE_KEY"
    text content "redacted text; NOT NULL; retention blanks it to a removal notice, never NULL"
    jsonb raw_payload "allowlisted fields, PII fields removed"
    text content_hash "SHA-256 of stored text; integrity hash, not a join key"
    text location "city level or empty"
    text language
    timestamptz collected_at
    uuid pseudo_user_id FK "nullable, 006"
    text provenance_fingerprint "017: HMAC-SHA256"
    uuid ingest_mv_id FK "022: ingest version stored under"
    timestamptz text_removed_at "025: set when retention blanks the text"
    text text_removed_reason "025"
    uuid admission_mv_id FK "042: admission_filter version that admitted it"
  }

  decision_audit_log {
    uuid id PK
    uuid raw_post_id FK
    uuid job_id FK
    uuid methodology_version_id FK
    text decision_type "sentiment | relevance | discourse"
    text model_name
    text input_hash "SHA-256 of the input; API serves HMAC"
    jsonb output
    real confidence
    timestamptz created_at
  }

  methodology_versions {
    uuid id PK
    text component "sentiment | relevance | discourse | bias | ingest | audit_narration | embedding | admission_filter"
    text version "UNIQUE with component"
    text model_name
    jsonb config
    text justification
    timestamptz effective_from
    timestamptz deprecated_at
  }

  sentiment_results {
    uuid id PK
    uuid raw_post_id FK
    uuid audit_id FK
    real score
    real comparative
    text indicator "positive | neutral | negative"
    text_array positive_words
    text_array negative_words
    int token_count
    timestamptz created_at
  }

  relevance_results {
    uuid id PK
    uuid raw_post_id FK
    uuid audit_id FK
    real score
    text_array matched_keywords
    boolean is_relevant
    timestamptz created_at
  }

  discourse_results {
    uuid id PK
    uuid raw_post_id FK
    uuid audit_id FK
    real dqi_total
    jsonb dimensions
    text argument_cluster_id
    real novelty_score
    timestamptz created_at
  }

  post_embeddings {
    uuid id PK
    uuid raw_post_id FK "UNIQUE"
    vector384 embedding "VECTOR(384), HNSW cosine index"
    text model_name
    text methodology_version "012"
    timestamptz created_at
  }

  bias_assessments {
    uuid id PK
    uuid job_id FK
    text assessment_type
    text group_field
    text group_value
    text metric_name
    real metric_value
    real threshold
    boolean is_violation
    text severity
    jsonb evidence
    uuid methodology_version_id FK "010: lineage"
    timestamptz created_at
  }

  data_sources {
    uuid id PK
  }

  processing_jobs {
    uuid id PK
  }
