  data_retention_log {
    uuid id PK
    uuid raw_post_id "nullable, not enforced"
    text action "compacted | purged_demo | blanked_platform_terms"
    text reason
    text legal_basis
    timestamptz performed_at
    text performed_by
  }

  monthly_topic_rollups {
    uuid id PK
    date rollup_month
    text topic_label
    text source_category "null = all"
    text location "null = global"
    text language "null = all"
    int post_count
    int positive_count
    int neutral_count
    int negative_count
    real avg_comparative
    real avg_dqi_score
    text_array top_keywords
    timestamptz created_at
  }

  monthly_source_rollups {
    uuid id PK
    date rollup_month "UNIQUE with source_id"
    uuid source_id FK
    text source_category
    int post_count
    int positive_count
    int neutral_count
    int negative_count
    real avg_comparative
    real avg_dqi_score
    timestamptz created_at
  }

  compaction_log {
    uuid id PK
    date compacted_month UK
    int posts_compacted
    int rollups_created
    int embeddings_deleted
    int content_nulled
    timestamptz completed_at
  }

  pseudonymous_users {
    uuid id PK
    text pseudo_id UK "verb-noun id"
    text style_cluster_id
    text_array topic_affinity
    int platform_count
    real correlation_confidence
    timestamptz first_sighted_at
    timestamptz last_sighted_at
  }

  user_platform_sightings {
    uuid id PK
    uuid pseudo_user_id FK
    uuid source_id FK
    text signal_hash
    real confidence
    timestamptz sighted_at
  }

  schema_migrations {
    text filename PK "created by scripts/migrate.js"
    timestamptz applied_at
  }

  raw_posts {
    uuid id PK
    uuid source_id FK
    uuid pseudo_user_id FK
  }

  data_sources {
    uuid id PK
  }

  data_retention_log }o..o| raw_posts : "raw_post_id"
  data_sources |o--o{ monthly_source_rollups : "source_id"
  pseudonymous_users ||--o{ user_platform_sightings : "pseudo_user_id"
  data_sources ||--o{ user_platform_sightings : "source_id"
  pseudonymous_users |o--o{ raw_posts : "pseudo_user_id"
  data_sources ||--o{ raw_posts : "source_id"
