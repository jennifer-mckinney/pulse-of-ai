  class PulseDesignConfig {
    <<config/design.config.js · data only>>
    THEMES, DEFAULT_THEME
    SENTIMENT_PALETTE, SENTIMENT_BUCKETS
    CATEGORIES 8, CATEGORY_SLUGS, CAT_COLORS, CAT_LABELS
    SEVERITY_COLORS
    GLOBE
  }
  class PulseApiConfig {
    <<config/api.config.js · data only>>
    ENDPOINTS 11 paths
    REFRESH_MS 150000
    CITY_POSTS_LIMIT 3
    TIMESERIES_HOURS 12
  }
  class PulseStoryConfig {
    <<config/story.config.js · data only>>
    STORY 11 beats
  }
  class PulseCityRegistry {
    <<config/cities.config.js · also required by the server>>
    CITY_REGISTRY
    +findCity(name)
    +launchCities() 30 tier-1 cities
  }
  class PulseLegalConfig {
    <<config/legal.config.js · data only>>
    UPSTREAM_URL
    SOURCE_URL
    NOTICE 6 lines
  }
  class PulseUtils {
    <<utils.js>>
    +esc(s)
    +healthState(health) state, label, systemAlerts
    +catLabel(slug)
    +netSentiment() · sentimentBucket()
    +fmtPct() · fmtCount() · fmtNet()
  }
  class PulseData {
    <<data.js>>
    DEMO_DATA
    +buildDemoData(cities)
    +normalizeCities(raw) keeps publisher_posts
    +mergeWithBaseline(rows)
    +loadCityData() cities, isDemo, dataMode
    +dataModeOf(rows) demo, live, mixed, none
    +labelsAsDemo(mode)
  }
  class PulseInsights {
    <<insights.js>>
    +computeInsights(cities)
    +renderTemplate(template, values)
    +ribbonRows() · allCategoryRows()
    +partitionThemes(themes, mode)
    +widestCategoryDivide() · catBreakdown()
  }
  class PulseChapters {
    <<chapters.js>>
    STORY · FALLBACK_COPY · INTRO_KICKERS
    +resolveChapter(beat, ins, cities, opts)
    +resolveIntro(ins, cities, dataMode) · introFacts() · introByline()
  }
  class PulseGlobe {
    <<globe.js · Canvas-2D>>
    +create(mount, deps) instance
    +init() · getInstance()
    +loadLandDots(url) vendor/world-atlas
    math pure namespace: publisherCount(), publisherLayer()
    publisher-location layer: dashed ring
    instance.setState(partial) · getState() · destroy()
    callbacks onHover, onDrag, onCityClick
    mark pulse:first-frame
  }
  class PulseStory {
    <<story.js>>
    +init()
    +getState() prog, exploring, activeIndex, isDemo, dataMode
    +getCities()
    +consumePendingCity()
    +setExploreSelection(cityId)
    fires pulse:exploring-changed, pulse:drill
    fires pulse:trace, pulse:data
    legend key: publisher location
  }
  class PulseUI {
    <<ui.js>>
    +init()
    +openAudit(post) audit drawer, 4 audiences
    +openHealth() health drawer, SYSTEM WATCHDOG section
    healthBanner() SYSTEM ALERT banner, watchdogModel()
    publisherTipText() tooltip line
    explore filters, city detail, tooltip, source ribbon
    listens to the four pulse:* events
  }
  class PulseMain {
    <<main.js · page shell>>
    +freezeInsightTimer()
    +scrollToExplore()
    sizes the scroll spacer, polls the health chip
    health chip h-system on a watchdog alert
    renders the about panel
    starts PulseGlobe, PulseStory and PulseUI
  }

  class Endpoints {
    <<called by the page>>
    data.js - GET /api/posts/aggregated-by-location
    story.js - GET /api/themes, GET /api/sources
    story.js - POST /api/query
    ui.js - POST /api/query
    ui.js - GET /api/sources/timeseries
    ui.js - GET /api/audit/POST_ID
    ui.js - GET /api/health, /api/bias/history
    ui.js - GET /api/methodology, /api/sources
    main.js - GET /api/health
    in ENDPOINTS, never called - /api/bias/latest, /api/refresh
  }
