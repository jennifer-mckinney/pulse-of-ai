  class Queues {
    <<queues/index.js · BullMQ on Valkey>>
    collect.rss, collect.api, collect.bulk attempts 1
    collect.refresh attempts 1
    ingest attempts 5, backoff 1 s
    embed attempts 5, backoff 2 s
    correlate attempts 5
    maintenance attempts 1
  }
  class WorkerProcess {
    <<workers/start.js>>
    concurrency: collect.rss 4, collect.api 4
    concurrency: collect.bulk 1, collect.refresh 1
    concurrency: ingest 8, embed 4, correlate 8
    concurrency: maintenance 1
    30 s tick: cycles, source health
    30 s tick: retention overdue, unscored sweep
    +schedule() every 10 min, also scheduleMaintenance
    +closeDueCycles() every 30 s
    +redditMaintenance() every 5 min
    +startHeartbeat() every 15 s
    +shutdown() on SIGTERM / SIGINT
  }
  class collectWorker {
    <<workers/collect.worker.js>>
    +processCollectJob(job) run summary, scoreVia queue
    +processRefreshJob(job) run summary, scoreVia queue
    +refreshDeadlineMs(env)
  }
  class scheduler {
    <<workers/collector.scheduler.js>>
    +scheduleAllSources(o: env, log) count
    each run also records gate + correlation gate events
  }
  class maintenanceWorker {
    <<workers/maintenance.worker.js>>
    retention every 5 min: text retention, stale jobs
    daily every 24 h: compaction, source_runs rollup, admission_rule_hits expiry, bias_window
    terms every 7 days: terms snapshots
    +processMaintenanceJob(job, o) steps
    +scheduleMaintenance(queue, env)
    +recordMaintenanceRun(task, out) maintenance_state
  }
  class ingestWorker {

  class heartbeat {
    <<workers/heartbeat.js + healthcheck.js>>
